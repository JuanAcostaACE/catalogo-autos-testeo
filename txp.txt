import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;

public class GuardarArchivo {
    public static void main(String[] args) {
        // 1. Define la ruta y el nombre del archivo (con la extensión que quieras, ej: .xtp)
        Path ruta = Paths.get("documento.xtp");
        
        // El contenido que deseas guardar
        String contenido = "Hola, esta es la línea 1 del examen.\nLínea 2 de datos.";

        try {
            // 2. Escribe el contenido en el archivo en una sola línea
            Files.writeString(ruta, contenido);
            System.out.println("¡Archivo guardado con éxito!");
            
        } catch (IOException e) {
            // 3. Manejo obligatorio de excepciones de entrada/salida
            System.out.println("Error al guardar el archivo: " + e.getMessage());
        }
    }
}
